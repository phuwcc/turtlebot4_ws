cmake_minimum_required(VERSION 3.8)
project(tb4_multi_bringup)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(fastrtps REQUIRED)
foreach(dependency rclcpp rclcpp_action geometry_msgs sensor_msgs nav_msgs nav2_msgs
                   tf2_msgs visualization_msgs std_srvs rcl_interfaces)
  find_package(${dependency} REQUIRED)
endforeach()
find_path(NLOHMANN_JSON_INCLUDE_DIR nlohmann/json.hpp)
if(NOT NLOHMANN_JSON_INCLUDE_DIR)
  message(FATAL_ERROR "Install nlohmann-json3-dev (bash scripts/setup.sh pc|robot)")
endif()
add_library(fleet_runtime STATIC src/fleet_runtime.cpp)
target_compile_features(fleet_runtime PUBLIC cxx_std_17)
target_include_directories(fleet_runtime PRIVATE ${NLOHMANN_JSON_INCLUDE_DIR})
ament_target_dependencies(fleet_runtime rclcpp rclcpp_action geometry_msgs sensor_msgs
  nav_msgs nav2_msgs tf2_msgs visualization_msgs std_srvs rcl_interfaces)
foreach(executable fleet_view socket_control)
  add_executable(${executable} src/fleet_main.cpp)
  target_link_libraries(${executable} fleet_runtime)
  ament_target_dependencies(${executable} rclcpp)
endforeach()
target_compile_definitions(fleet_view PRIVATE TB4_SOCKET_ENABLED=0)
target_compile_definitions(socket_control PRIVATE TB4_SOCKET_ENABLED=1)
add_executable(remote_velocity src/remote_velocity.cpp)
target_compile_features(remote_velocity PRIVATE cxx_std_17)
ament_target_dependencies(remote_velocity rclcpp geometry_msgs)
install(TARGETS fleet_view socket_control remote_velocity DESTINATION lib/${PROJECT_NAME})
add_executable(dds_graph_check src/dds_graph_check.cpp)
target_compile_features(dds_graph_check PRIVATE cxx_std_14)
target_link_libraries(dds_graph_check fastrtps)
install(TARGETS dds_graph_check DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(test_service_qos test/test_service_qos.cpp)
  target_compile_features(test_service_qos PRIVATE cxx_std_14)
  target_link_libraries(test_service_qos fastrtps)
  target_compile_definitions(test_service_qos PRIVATE TEST_SERVICE_PROFILE="${CMAKE_CURRENT_SOURCE_DIR}/config/common/fastdds_pc.xml")
  add_test(NAME service_qos COMMAND test_service_qos)
  add_executable(test_graph_wire test/test_graph_wire.cpp)
  target_compile_features(test_graph_wire PRIVATE cxx_std_14)
  add_test(NAME graph_wire COMMAND test_graph_wire)
endif()
ament_python_install_package(${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})
install(DIRECTORY ../../maps/ DESTINATION share/${PROJECT_NAME}/maps)
ament_package()
