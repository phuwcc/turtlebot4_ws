cmake_minimum_required(VERSION 3.8)
project(tb4_tools)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra -Wpedantic)
find_package(ament_cmake REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(rclcpp REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
add_executable(start_relative_mapping src/start_relative_mapping.cpp)
target_include_directories(start_relative_mapping PRIVATE include)
ament_target_dependencies(start_relative_mapping ament_index_cpp rclcpp geometry_msgs nav_msgs tf2_ros)
add_executable(teleop_keyboard src/teleop_keyboard.cpp)
target_include_directories(teleop_keyboard PRIVATE include)
ament_target_dependencies(teleop_keyboard rclcpp geometry_msgs)
install(TARGETS start_relative_mapping teleop_keyboard DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY config DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  enable_testing()
  add_executable(test_controls test/test_controls.cpp)
  target_include_directories(test_controls PRIVATE include)
  add_test(NAME controls COMMAND test_controls)
endif()
ament_package()
